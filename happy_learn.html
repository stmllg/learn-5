function doGet(e) { return processOffering(e); } function doPost(e) { return processOffering(e); } function processOffering(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var params = (e && e.parameter) ? e.parameter : {}; var classId = params.classId ? String(params.classId).trim() : ''; var userName = params.userName ? String(params.userName).trim() : ''; var coins = parseInt(params.coins, 10) || 0; var callback = params.callback || ''; var now = new Date(); // 使用台北時區格式化今天的日期字串 YYYY/MM/DD var todayStr = Utilities.formatDate(now, "Asia/Taipei", "yyyy/MM/dd"); // 寫入本次供養 (A時間戳記, B班號, C姓名, D供養福慧幣) if (userName && coins > 0) { sheet.appendRow([now, classId, userName, coins]); SpreadsheetApp.flush(); } // 讀取試算表資料進行加總 var data = sheet.getDataRange().getValues(); var todayMap = {}; var totalMap = {}; for (var i = 1; i < data.length; i++) { var row = data[i]; if (!row || row.length < 4) continue; var rawTime = row[0]; // A 欄：時間戳記 var rClass = row[1] ? String(row[1]).trim() : ''; // B 欄：班號 var rName = row[2] ? String(row[2]).trim() : ''; // C 欄：姓名 var rCoins = Number(row[3]) || 0; // D 欄：福慧幣 if (!rName || rCoins <= 0) continue; // 解析該列的日期字串 (轉為 yyyy/MM/dd) var rowDateStr = ""; if (rawTime instanceof Date) { rowDateStr = Utilities.formatDate(rawTime, "Asia/Taipei", "yyyy/MM/dd"); } else { var str = String(rawTime); var match = str.match(/(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})/); if (match) { var y = match[1]; var m = String(match[2]).padStart(2, '0'); var d = String(match[3]).padStart(2, '0'); rowDateStr = y + '/' + m + '/' + d; } } var key = rClass ? (rClass + '_' + rName) : rName; // 1. 總榜累計 if (!totalMap[key]) { totalMap[key] = { classId: rClass, name: rName, coins: 0 }; } totalMap[key].coins += rCoins; // 2. 當日榜累計 if (rowDateStr === todayStr) { if (!todayMap[key]) { todayMap[key] = { classId: rClass, name: rName, coins: 0 }; } todayMap[key].coins += rCoins; } } // 姓名去識別化 function maskUserName(name) { if (!name) return ''; var str = String(name).trim(); if (str.length <= 1) return str; if (str.length === 2) return str[0] + '*'; return str[0] + '*'.repeat(str.length - 2) + str[str.length - 1]; } function extractTop3(mapObj) { var list = []; for (var k in mapObj) { list.push(mapObj[k]); } list.sort(function(a, b) { return b.coins - a.coins; }); return list.slice(0, 3).map(function(item) { return { classId: item.classId, maskName: maskUserName(item.name), coins: item.coins }; }); } var result = { status: 'success', todayTop3: extractTop3(todayMap), totalTop3: extractTop3(totalMap) }; var jsonText = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + '(' + jsonText + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonText) .setMimeType(ContentService.MimeType.JSON); } } catch (err) { var errObj = JSON.stringify({ status: 'error', message: err.toString() }); if (callback) { return ContentService.createTextOutput(callback + '(' + errObj + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(errObj).setMimeType(ContentService.MimeType.JSON); } }
